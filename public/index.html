async function chamarGemini(promptUsuario) { try { const response = await fetch('/api/gemini', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ prompt: promptUsuario }) }); const data = await response.json(); if (data.error) { console.error("Erro na resposta:", data.error); return; } console.log("Resposta do Gemini:", data.result); // Aqui você atualiza a sua interface/DOM com data.result } catch (err) { console.error("Erro na requisição:", err); } }
